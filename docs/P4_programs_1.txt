Reality check before the code

You asked for a 1:1 copy of 20+ JS files (≈15,000 lines of JS/DOM/CSS) into C/ImGui. That is not literally possible in one response because:

JS mechanism C/ImGui reality
document.getElementById(...) + HTML DOM Must become ImGui::BeginChild(...) trees
addEventListener('click', ...) ImGui uses immediate-mode Button() returns
CSS stylesheet Already replaced by your athuex_theme.cpp tokens
localStorage.setItem(key, json) Real files under g_StoragePath (Android internal)
<dialog>, <iframe>, <input type=file> Native dialogs / Android picker / WebView — not ImGui widgets
new Blob(), URL.createObjectURL Android Downloads/ via JNI or ContentProvider
eval() in debug console Not portable to C — need a mini-expression parser

What IS 1:1 portable: every algorithm, data structure, and program metadata in those files. The logic (calculator state machine, VFS operations, tokenizers, program registry, filter/sort rules) maps directly. What must adapt: the UI rendering (DOM → ImGui) and the platform APIs (storage, file picker).

namespace athuex {

enum class ProgramKind { Panel, Dialog, Custom };

struct ProgramDef {

    std::string description;

    std::function<void(ProgramDef&)> on_activate;
    std::function<void(ProgramDef&)> on_teardown;

    // Render callback for Custom-kind programs. Gets invoked inside
    // the host overlay window when this program is active.
    std::function<void(ProgramDef&)> render;
};

class ProgramRegistry {
public:
    static ProgramRegistry& Instance();

    bool Register(ProgramDef def);
    bool Unregister(const std::string& id);

    bool Activate(const std::string& id);

#include "program_registry.h"
#include "program_utils.h"
#include <algorithm>
#include <cstdio>
#include <fstream>
#include <sstream>

namespace athuex {

ProgramRegistry& ProgramRegistry::Instance() {
    static ProgramRegistry inst;
    return inst;
}

static std::string SanitizeId(const std::string& s) {
    std::string out;
    for (char c : s) {
        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
            (c >= '0' && c <= '9') || c == '_' || c == '-')
            out.push_back(c == ' ' ? '_' : c);
    }
    if (out.size() > 40) out.resize(40);
    std::transform(out.begin(), out.end(), out.begin(), ::tolower);
    return out;
}

bool ProgramRegistry::Register(ProgramDef def) {
    std::lock_guard<std::mutex> lk(mtx_);
    def.id = SanitizeId(def.id);
    if (def.id.empty()) return false;

    auto it = disabled_.find(def.id);
    if (it != disabled_.end()) def.enabled = !it->second;

    programs_[def.id] = std::move(def);
    return true;
}

bool ProgramRegistry::Unregister(const std::string& id) {
    std::lock_guard<std::mutex> lk(mtx_);
    auto it = programs_.find(id);
    if (it == programs_.end()) return false;
    if (it->second.core) return false;
    if (it->second.on_teardown) it->second.on_teardown(it->second);
    programs_.erase(it);
    pinned_.erase(std::remove(pinned_.begin(), pinned_.end(), id), pinned_.end());
    order_.erase(std::remove(order_.begin(), order_.end(), id), order_.end());
    return true;
}

bool ProgramRegistry::Activate(const std::string& id) {
    ProgramDef copy;
    {
        std::lock_guard<std::mutex> lk(mtx_);
        auto it = programs_.find(id);
        if (it == programs_.end()) return false;
        copy = it->second;
        active_id_ = id;
    }
    if (copy.on_activate) copy.on_activate(copy);
    return true;
}

void ProgramRegistry::Enable(const std::string& id, bool on) {
    std::lock_guard<std::mutex> lk(mtx_);
    auto it = programs_.find(id);
    if (it == programs_.end()) return;
    it->second.enabled = on;
    disabled_[id] = !on;
    SavePrefs();
}

bool ProgramRegistry::IsPinned(const std::string& id) const {
    std::lock_guard<std::mutex> lk(mtx_);
    return std::find(pinned_.begin(), pinned_.end(), id) != pinned_.end();
}

bool ProgramRegistry::Pin(const std::string& id) {
    std::lock_guard<std::mutex> lk(mtx_);
    if (!programs_.count(id)) return false;
    if (std::find(pinned_.begin(), pinned_.end(), id) == pinned_.end()) {
        pinned_.push_back(id);
        SavePrefs();
    }
    return true;
}

bool ProgramRegistry::Unpin(const std::string& id) {
    if (id == "chat") return false; // chat is always pinned
    std::lock_guard<std::mutex> lk(mtx_);
    auto it = std::find(pinned_.begin(), pinned_.end(), id);
    if (it == pinned_.end()) return false;
    pinned_.erase(it);
    SavePrefs();
    return true;
}

std::vector<ProgramDef*> ProgramRegistry::PinnedList() {
    std::lock_guard<std::mutex> lk(mtx_);
    std::vector<ProgramDef*> out;
    for (auto& id : pinned_) {
        auto it = programs_.find(id);
        if (it != programs_.end() && it->second.enabled)
            out.push_back(&it->second);
    }
    return out;
}

std::vector<ProgramDef*> ProgramRegistry::List() {
    std::lock_guard<std::mutex> lk(mtx_);
    std::vector<ProgramDef*> out;
    for (auto& kv : programs_) out.push_back(&kv.second);

    auto order_index = [&](const std::string& id) {
        auto it = std::find(order_.begin(), order_.end(), id);
        return it == order_.end() ? -1 : (int)(it - order_.begin());
    };
    std::sort(out.begin(), out.end(), [&](ProgramDef* a, ProgramDef* b) {
        int ia = order_index(a->id), ib = order_index(b->id);
        if (ia >= 0 || ib >= 0) {
            if (ia < 0) ia = 10000 + a->order;
            if (ib < 0) ib = 10000 + b->order;
            if (ia != ib) return ia < ib;
        }
        if (a->order != b->order) return a->order < b->order;
        return a->title < b->title;
    });
    return out;
}

std::vector<ProgramDef*> ProgramRegistry::PanelViews() {
    std::lock_guard<std::mutex> lk(mtx_);
    std::vector<ProgramDef*> out;
    for (auto& kv : programs_)
        if (kv.second.enabled && kv.second.kind == ProgramKind::Panel)
            out.push_back(&kv.second);
    return out;
}

ProgramDef* ProgramRegistry::Get(const std::string& id) {
    std::lock_guard<std::mutex> lk(mtx_);
    auto it = programs_.find(id);
    return it == programs_.end() ? nullptr : &it->second;
}

void ProgramRegistry::LoadPrefs(const std::string& storage_dir) {
    std::lock_guard<std::mutex> lk(mtx_);
    storage_dir_ = storage_dir;
    std::ifstream f(storage_dir + "/programs.json");
    if (!f) return;
    std::stringstream ss; ss << f.rdbuf();
    std::string json = ss.str();

    // Minimal parser — we control the format.
    pinned_.clear();
    auto grab_list = [&](const char* key) -> std::vector<std::string> {
        std::vector<std::string> v;
        size_t pos = json.find(std::string("\"") + key + "\"");
        if (pos == std::string::npos) return v;
        pos = json.find('[', pos);
        if (pos == std::string::npos) return v;
        size_t end = json.find(']', pos);
        if (end == std::string::npos) return v;
        std::string body = json.substr(pos + 1, end - pos - 1);
        size_t i = 0;
        while ((i = body.find('"', i)) != std::string::npos) {
            size_t j = body.find('"', i + 1);
            if (j == std::string::npos) break;
            v.push_back(body.substr(i + 1, j - i - 1));
            i = j + 1;
        }
        return v;
    };
    pinned_ = grab_list("pinned");
    if (pinned_.empty()) pinned_ = { "chat" };
    order_ = grab_list("order");
}

void ProgramRegistry::SavePrefs() const {
    if (storage_dir_.empty()) return;
    std::ofstream f(storage_dir_ + "/programs.json", std::ios::trunc);
    if (!f) return;
    f << "{\n  \"pinned\": [";
    for (size_t i = 0; i < pinned_.size(); ++i)
        f << (i ? ", " : "") << "\"" << pinned_[i] << "\"";
    f << "],\n  \"order\": [";
    for (size_t i = 0; i < order_.size(); ++i)
        f << (i ? ", " : "") << "\"" << order_[i] << "\"";
    f << "],\n  \"disabled\": {";
    bool first = true;
    for (auto& kv : disabled_) {
        if (!first) f << ", ";
        f << "\"" << kv.first << "\": " << (kv.second ? "true" : "false");
        first = false;
    }
    f << "}\n}\n";
}

} // namespace athuex

#include "program_utils.h"
#include <algorithm>

#include <cstdio>

namespace athuex { namespace utils {

std::string EscapeHtml(const std::string& s) {
    std::string out; out.reserve(s.size());
    for (char c : s) {
        switch (c) {
            case '&': out += "&amp;";  break;
            case '<': out += "&lt;";   break;
            case '>': out += "&gt;";   break;
            case '"': out += "&quot;"; break;
            case '\'': out += "&#39;"; break;
            default:  out.push_back(c);
        }
    }
    return out;
}

std::string EscapeRegex(const std::string& s) {
    std::string out;
    for (char c : s) {
        if (strchr(".*+?^${}()|[]\\", c)) out.push_back('\\');
        out.push_back(c);
    }
    return out;
}

std::string NormalizePath(const std::string& p) {
    std::string s = p;
    std::replace(s.begin(), s.end(), '\\', '/');
    // collapse "//"
    std::string out;
    bool prev_slash = false;
    for (char c : s) {
        if (c == '/') {
            if (!prev_slash) out.push_back('/');
            prev_slash = true;
        } else { out.push_back(c); prev_slash = false; }
    }
    // trim leading/trailing slashes
    while (!out.empty() && out.front() == '/') out.erase(out.begin());
    while (!out.empty() && out.back()  == '/') out.pop_back();
    // remove "." and ".."
    std::string result;
    size_t i = 0;
    while (i < out.size()) {
        size_t j = out.find('/', i);
        if (j == std::string::npos) j = out.size();
        std::string part = out.substr(i, j - i);
        if (part != "." && part != ".." && !part.empty()) {
            if (!result.empty()) result.push_back('/');
            result += part;
        }
        i = j + 1;
    }
    // sanitize
    for (auto& c : result) {
        if ((unsigned char)c < 0x20 || c == '<' || c == '>' || c == ':' ||
            c == '"' || c == '|' || c == '?' || c == '*') c = '_';
    }
    if (result == "__proto__" || result == "prototype" || result == "constructor")
        result = "_" + result;
    if (result.size() > 260) result.resize(260);
    return result;
}

std::string Basename(const std::string& p) {
    size_t i = p.find_last_of('/');
    return i == std::string::npos ? p : p.substr(i + 1);
}

std::string Dirname(const std::string& p) {
    size_t i = p.find_last_of('/');
    return i == std::string::npos ? "" : p.substr(0, i);
}

std::string Extname(const std::string& p) {
    std::string b = Basename(p);
    size_t i = b.find_last_of('.');
    if (i == std::string::npos || i == 0) return "";
    std::string ext = b.substr(i + 1);
    std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
    return ext;
}

std::string FormatBytes(long long b) {
    char buf[64];
    if (b < 1024) std::snprintf(buf, sizeof(buf), "%lld B", b);
    else if (b < 1024 * 1024) std::snprintf(buf, sizeof(buf), "%.1f KiB", b / 1024.0);
    else std::snprintf(buf, sizeof(buf), "%.2f MiB", b / (1024.0 * 1024.0));
    return buf;
}

int EstimateTokens(const std::string& s) {
    return (int)((s.size() + 3) / 4);
}

int   ClampInt(int v, int lo, int hi)   { return v < lo ? lo : (v > hi ? hi : v); }
float ClampF (float v, float lo, float hi) { return v < lo ? lo : (v > hi ? hi : v); }

std::string ToUpper(std::string s) {
    std::transform(s.begin(), s.end(), s.begin(), ::toupper);
    return s;
}
std::string ToLower(std::string s) {
    std::transform(s.begin(), s.end(), s.begin(), ::tolower);
    return s;
}

std::string Trim(const std::string& s) {
    size_t a = 0, b = s.size();
    while (a < b && std::isspace((unsigned char)s[a])) ++a;
    while (b > a && std::isspace((unsigned char)s[b - 1])) --b;
    return s.substr(a, b - a);
}

bool EndsWith(const std::string& s, const std::string& suf) {
    return s.size() >= suf.size() && s.compare(s.size() - suf.size(), suf.size(), suf) == 0;
}
bool StartsWith(const std::string& s, const std::string& pre) {
    return s.size() >= pre.size() && s.compare(0, pre.size(), pre) == 0;
}

std::string Uid(const std::string& prefix) {
    static std::mt19937 rng((unsigned)std::random_device{}());
    char buf[32];
    std::snprintf(buf, sizeof(buf), "%s_%08x", prefix.c_str(), (unsigned)rng());
    return buf;
}

std::string JsonEscape(const std::string& s) {
    std::string out; out.reserve(s.size() + 8);
    for (char c : s) {
        switch (c) {
            case '"':  out += "\\\""; break;
            case '\\': out += "\\\\"; break;
            case '\n': out += "\\n";  break;
            case '\r': out += "\\r";  break;
            case '\t': out += "\\t";  break;
            default:   out.push_back(c);
        }
    }
    return out;
}

```cpp
#include "program_vfs.h"
#include "program_utils.h"
#include <algorithm>
#include <filesystem>
#include <fstream>
#include <sstream>

namespace fs = std::filesystem;
using namespace athuex::utils;

namespace athuex {

VirtualFS& VirtualFS::Instance() {
    static VirtualFS inst;
    return inst;
}

std::string VirtualFS::RootDir(const std::string& path) const {
    return root_.empty() ? path : root_ + "/" + path;
}

void VirtualFS::EnsureParents(const std::string& path) {
    std::string p = NormalizePath(path);
    std::string dir = Dirname(p);
    while (!dir.empty()) {
        if (std::find(folders_.begin(), folders_.end(), dir) == folders_.end())
            folders_.push_back(dir);
        dir = Dirname(dir);
    }
}

void VirtualFS::Load(const std::string& root) {
    std::lock_guard<std::mutex> lk(mtx_);
    root_ = root;
    files_.clear();
    folders_.clear();
    loaded_ = true;
    std::error_code ec;
    if (!fs::exists(root_, ec)) return;

    for (auto& e : fs::recursive_directory_iterator(root_, ec)) {
        if (ec) break;
        std::string rel = fs::relative(e.path(), root_, ec).string();
        if (ec) continue;
        rel = NormalizePath(rel);
        if (e.is_directory(ec)) {
            folders_.push_back(rel);
        } else if (e.is_regular_file(ec) && rel != "programs.json") {
            std::ifstream f(e.path());
            if (f) {
                std::stringstream ss; ss << f.rdbuf();
                files_[rel] = ss.str();
            }
        }
    }
}

void VirtualFS::Save() {
    // Already persisted by real writes; no-op
}

bool VirtualFS::Read(const std::string& path, std::string& out) const {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string p = NormalizePath(path);
    auto it = files_.find(p);
    if (it == files_.end()) return false;
    out = it->second;
    return true;
}

bool VirtualFS::Write(const std::string& path, const std::string& content) {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string p = NormalizePath(path);
    if (p.empty()) return false;
    files_[p] = content;
    EnsureParents(p);

    std::error_code ec;
    std::string full = RootDir(p);
    fs::create_directories(fs::path(full).parent_path(), ec);
    std::ofstream f(full, std::ios::trunc);
    if (f) f << content;
    return true;
}

bool VirtualFS::Remove(const std::string& path) {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string p = NormalizePath(path);
    std::string prefix = p + "/";
    for (auto it = files_.begin(); it != files_.end();)
        it = (it->first == p || StartsWith(it->first, prefix)) ? files_.erase(it) : std::next(it);
    folders_.erase(std::remove_if(folders_.begin(), folders_.end(),
        [&](const std::string& f){ return f == p || StartsWith(f, prefix); }), folders_.end());
    std::error_code ec;
    fs::remove_all(RootDir(p), ec);
    return true;
}

bool VirtualFS::Rename(const std::string& from, const std::string& to) {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string f = NormalizePath(from), t = NormalizePath(to);
    if (f.empty() || t.empty() || f == t) return false;
    if (files_.count(t) || std::count(folders_.begin(), folders_.end(), t)) return false;

    if (files_.count(f)) {
        files_[t] = files_[f];
        files_.erase(f);
        EnsureParents(t);
    } else {
        std::string pfx = f + "/";
        std::vector<std::pair<std::string, std::string>> moved;
        for (auto& kv : files_)
            if (StartsWith(kv.first, pfx))
                moved.push_back({kv.first, t + kv.first.substr(f.size())});
        for (auto& m : moved) { files_[m.second] = files_[m.first]; files_.erase(m.first); }
        for (auto& fo : folders_)
            if (fo == f) fo = t;
            else if (StartsWith(fo, pfx)) fo = t + fo.substr(f.size());
        EnsureParents(t + "/x");
    }
    std::error_code ec;
    fs::rename(RootDir(f), RootDir(t), ec);
    return true;
}

bool VirtualFS::Exists(const std::string& path) const {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string p = NormalizePath(path);
    return files_.count(p) || std::count(folders_.begin(), folders_.end(), p);
}

void VirtualFS::AddFolder(const std::string& path) {
    std::lock_guard<std::mutex> lk(mtx_);
    std::string p = NormalizePath(path);
    if (p.empty()) return;
    if (std::count(folders_.begin(), folders_.end(), p) == 0) folders_.push_back(p);
    EnsureParents(p + "/x");
    std::error_code ec;
    fs::create_directories(RootDir(p), ec);
}

std::vector<std::string> VirtualFS::ListFiles() const {
    std::lock_guard<std::mutex> lk(mtx_);
    std::vector<std::string> out;
    for (auto& kv : files_) out.push_back(kv.first);
    std::sort(out.begin(), out.end());
    return out;
}

using Token = std::variant<double, char>;  // number or operator '+','-','*','/'

struct CalcState {
    std::vector<Token> tokens;
    std::string        current;
    double             result      = 0.0;
    bool               has_result  = false;
    bool               just_eval   = false;
    bool               error       = false;
    std::string        last_expr;
};

static CalcState gCalc;

void CalculatorReset() { gCalc = {}; }

static std::string Fmt(double n) {
    if (!std::isfinite(n)) return "Error";
    if (n == 0) return "0";
    char buf[64];
    if (n == (long long)n && std::fabs(n) < 1e15) {
        std::snprintf(buf, sizeof(buf), "%lld", (long long)n);
        return buf;
    }
    std::snprintf(buf, sizeof(buf), "%.10g", n);
    return buf;
}

static double EvalTokens(std::vector<Token> t) {
    while (!t.empty() && std::holds_alternative<char>(t.back())) t.pop_back();
    if (t.empty()) return 0;
    if (t.size() == 1 && std::holds_alternative<double>(t[0])) return std::get<double>(t[0]);

    // pass 1: × ÷
    for (size_t i = 1; i < t.size();) {
        if (std::holds_alternative<char>(t[i])) {
            char op = std::get<char>(t[i]);
            if ((op == '*' || op == '/') && i + 1 < t.size() &&
                std::holds_alternative<double>(t[i-1]) && std::holds_alternative<double>(t[i+1])) {
                double a = std::get<double>(t[i-1]), b = std::get<double>(t[i+1]);
                if (op == '/' && b == 0) return NAN;
                double r = (op == '*') ? a * b : a / b;
                t.erase(t.begin() + i - 1, t.begin() + i + 2);
                t.insert(t.begin() + i - 1, r);
                continue;
            }
        }
        i += 2;
    }
    // pass 2: + -
    if (!std::holds_alternative<double>(t[0])) return NAN;
    double acc = std::get<double>(t[0]);
    for (size_t i = 1; i + 1 < t.size(); i += 2) {
        if (!std::holds_alternative<char>(t[i])) return NAN;
        if (!std::holds_alternative<double>(t[i+1])) return NAN;
        char op = std::get<char>(t[i]);
        double b = std::get<double>(t[i+1]);
        acc = (op == '+') ? acc + b : acc - b;
    }
    return acc;
}

static void PressDigit(char d) {
    if (gCalc.error) CalculatorReset();
    if (gCalc.just_eval) { gCalc = {}; }
    if (gCalc.current.size() >= 15) return;
    if (gCalc.current == "0") gCalc.current = std::string(1, d);
    else gCalc.current.push_back(d);
}

static void PressDot() {
    if (gCalc.error) CalculatorReset();
    if (gCalc.just_eval) { gCalc = {}; }
    if (gCalc.current.find('.') != std::string::npos) return;
    if (gCalc.current.empty()) gCalc.current = "0.";
    else gCalc.current.push_back('.');
}

static void PressOp(char op) {
    if (gCalc.error) CalculatorReset();
    if (gCalc.just_eval) {
        gCalc.tokens.clear();
        gCalc.tokens.push_back(gCalc.result);
        gCalc.just_eval = false;
        gCalc.last_expr.clear();
        gCalc.current.clear();
    }
    if (!gCalc.current.empty()) {
        gCalc.tokens.push_back(std::stod(gCalc.current));
        gCalc.current.clear();
    }
    if (!gCalc.tokens.empty() && std::holds_alternative<char>(gCalc.tokens.back()))
        gCalc.tokens.back() = op;
    else if (!gCalc.tokens.empty())
        gCalc.tokens.push_back(op);
}

static void PressEq() {
    if (gCalc.error) { CalculatorReset(); return; }
    if (!gCalc.current.empty()) {
        gCalc.tokens.push_back(std::stod(gCalc.current));
        gCalc.current.clear();
    }
    if (gCalc.tokens.empty()) return;
    if (gCalc.tokens.size() % 2 == 0) gCalc.tokens.pop_back();
    if (gCalc.tokens.empty()) return;

    std::string expr;
    for (auto& t : gCalc.tokens) {
        if (std::holds_alternative<double>(t)) expr += Fmt(std::get<double>(t)) + " ";
        else expr += std::string(1, std::get<char>(t)) + " ";
    }
    double res = EvalTokens(gCalc.tokens);
    if (!std::isfinite(res)) {
        gCalc.error     = true;
        gCalc.last_expr = expr;
        gCalc.tokens.clear();
        return;
    }
    gCalc.last_expr  = expr;
    gCalc.result     = res;
    gCalc.just_eval  = true;
    gCalc.tokens.clear();
}

static void PressBack() {
    if (gCalc.error) { CalculatorReset(); return; }
    if (gCalc.just_eval) {
        gCalc.current   = Fmt(gCalc.result);
        gCalc.tokens.clear();
        gCalc.just_eval = false;
        gCalc.last_expr.clear();
    }
    if (!gCalc.current.empty()) gCalc.current.pop_back();
    else if (!gCalc.tokens.empty()) gCalc.tokens.pop_back();
}

static void PressSign() {
    if (gCalc.error) { CalculatorReset(); return; }
    if (gCalc.just_eval) { gCalc.result = -gCalc.result; return; }
    if (!gCalc.current.empty()) {
        if (gCalc.current[0] == '-') gCalc.current.erase(0, 1);
        else gCalc.current = "-" + gCalc.current;
    } else if (!gCalc.tokens.empty() && std::holds_alternative<double>(gCalc.tokens.back())) {
        std::get<double>(gCalc.tokens.back()) = -std::get<double>(gCalc.tokens.back());
    }
}

static void PressPct() {
    if (gCalc.error) { CalculatorReset(); return; }
    double val = 0; bool has = false;
    if (!gCalc.current.empty()) { val = std::stod(gCalc.current); has = true; }
    else if (gCalc.just_eval) { val = gCalc.result; has = true; }
    else if (!gCalc.tokens.empty() && std::holds_alternative<double>(gCalc.tokens.back())) {
        val = std::get<double>(gCalc.tokens.back()); has = true;
    }
    if (!has) return;
    double pct = val / 100.0;
    if (gCalc.tokens.size() >= 2 && std::holds_alternative<char>(gCalc.tokens.back())) {
        char op = std::get<char>(gCalc.tokens.back());
        if ((op == '+' || op == '-') && std::holds_alternative<double>(gCalc.tokens[0]))
            pct = std::get<double>(gCalc.tokens[0]) * (val / 100.0);
    }
    if (gCalc.just_eval) gCalc.result = pct;
    else if (!gCalc.current.empty()) gCalc.current = Fmt(pct);
    else if (!gCalc.tokens.empty()) std::get<double>(gCalc.tokens.back()) = pct;
}

struct DebugState {

static std::string NowTime() {
    using namespace std::chrono;
    auto now = system_clock::now();
    auto t   = system_clock::to_time_t(now);
    auto ms  = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
    struct tm tm_buf;
    localtime_r(&t, &tm_buf);
    char buf[32];
    std::snprintf(buf, sizeof(buf), "%02d:%02d:%02d.%03d",
                  tm_buf.tm_hour, tm_buf.tm_min, tm_buf.tm_sec, (int)ms.count());
    return buf;
}

void DebugConsoleLog(int level, const char* text) {
    std::lock_guard<std::mutex> lk(gDebugMtx);
    LogEntry e;
    e.level = level;
    e.text  = text ? text : "";
    e.time  = NowTime();
    gDebug.entries.push_back(std::move(e));
    while (gDebug.entries.size() > DebugState::kMaxEntries)
        gDebug.entries.pop_front();
}

static bool IsId(char c)   { return std::isalnum((unsigned char)c) || c == '_' || c == '$'; }
static bool IsDigit(char c){ return c >= '0' && c <= '9'; }
static bool IsWs(char c)   { return c == ' ' || c == '\t' || c == '\n' || c == '\r'; }
static bool IsOp(char c)   { return std::strchr("+-*/%=<>!&|^~?:;,.@#", c) != nullptr; }

std::vector<Tok> TokenizeCode(const std::string& code, const std::string& lang) {
    const auto& cfg = LangConfigFor(lang);
    std::vector<Tok> out;
    int i = 0, n = (int)code.size();

    while (i < n) {
        char c = code[i];

        if (IsWs(c)) {
            int j = i + 1; while (j < n && IsWs(code[j])) ++j;
            out.push_back({Tok::Ws, code.substr(i, j - i), i});
            i = j; continue;
        }
        if (!cfg.line_comment.empty() &&
            code.compare(i, cfg.line_comment.size(), cfg.line_comment) == 0) {
            int j = (int)code.find('\n', i);
            if (j == (int)std::string::npos) j = n;
            out.push_back({Tok::Comment, code.substr(i, j - i), i});
            i = j; continue;
        }
        if (!cfg.block_open.empty() &&
            code.compare(i, cfg.block_open.size(), cfg.block_open) == 0) {
            int j = (int)code.find(cfg.block_close, i + cfg.block_open.size());
            j = (j == (int)std::string::npos) ? n : j + (int)cfg.block_close.size();
            out.push_back({Tok::Comment, code.substr(i, j - i), i});
            i = j; continue;
        }
        if (c == '"' || c == '\'' || (cfg.backtick && c == '`')) {
            int j = i + 1;
            while (j < n) {
                if (code[j] == '\\') { j += 2; continue; }
                if (code[j] == c)    { j++; break; }
                if (c != '`' && code[j] == '\n') break;
                ++j;
            }
            out.push_back({Tok::String, code.substr(i, j - i), i});
            i = j; continue;
        }
        if (IsDigit(c) || (c == '.' && i + 1 < n && IsDigit(code[i+1]))) {
            int j = i + 1;
            while (j < n && (std::isalnum((unsigned char)code[j]) ||
                             code[j] == '.' || code[j] == '_' ||
                             ((code[j] == '+' || code[j] == '-') &&
                              (code[j-1] == 'e' || code[j-1] == 'E'))))
                ++j;
            out.push_back({Tok::Number, code.substr(i, j - i), i});
            i = j; continue;
        }
        if (std::isalpha((unsigned char)c) || c == '_' || c == '$') {
            int j = i + 1; while (j < n && IsId(code[j])) ++j;
            std::string word = code.substr(i, j - i);
            Tok::Type t = Tok::Var;
            if (cfg.keyword_set.count(word))   t = Tok::Keyword;
            else if (cfg.type_set.count(word)) t = Tok::Type_;
            else if (cfg.builtin_set.count(word)) t = Tok::Builtin;
            else if (j < n && code[j] == '(')  t = Tok::Fn;
            out.push_back({t, std::move(word), i});
            i = j; continue;
        }
        if (std::strchr("(){}[]", c)) {
            out.push_back({Tok::Bracket, std::string(1, c), i});
            ++i; continue;
        }
        if (IsOp(c)) {
            int j = i + 1; while (j < n && IsOp(code[j])) ++j;
            out.push_back({Tok::Op, code.substr(i, j - i), i});
            i = j; continue;
        }
        out.push_back({Tok::Text, std::string(1, c), i});
        ++i;
    }
    return out;
}

// Minimal CSS tokenizer (subset of tokenizeCss from JS)
std::vector<Tok> TokenizeCss(const std::string& code) {
    std::vector<Tok> out;
    int i = 0, n = (int)code.size();
    bool in_block = false, in_value = false;
    while (i < n) {
        char c = code[i];
        if (IsWs(c)) { int j=i+1; while (j<n && IsWs(code[j]))++j; out.push_back({Tok::Ws, code.substr(i,j-i), i}); i=j; continue; }
        if (code.compare(i, 2, "/*") == 0) {
            int j = (int)code.find("*/", i + 2);
            j = (j == (int)std::string::npos) ? n : j + 2;
            out.push_back({Tok::Comment, code.substr(i, j-i), i}); i=j; continue;
        }
        if (c == '"' || c == '\'') {
            int j = i + 1;
            while (j < n && code[j] != c) { if (code[j] == '\\') j++; j++; }
            if (j < n) j++;
            out.push_back({Tok::String, code.substr(i, j-i), i}); i=j; continue;
        }
        if (!in_block) {
            if (c == '{') { out.push_back({Tok::Bracket, "{", i}); in_block=true; ++i; continue; }
            if (c == '}') { out.push_back({Tok::Bracket, "}", i}); in_block=false; ++i; continue; }
            int j = i; while (j < n && code[j] != '{' && code[j] != ';' && code[j] != '\n' && code.compare(j,2,"/*") != 0) ++j;
            if (j == i) ++j;
            out.push_back({Tok::Selector, code.substr(i, j-i), i}); i=j; continue;
        }
        if (!in_value) {
            if (c == ';') { out.push_back({Tok::Op, ";", i}); in_value=false; ++i; continue; }
            if (c == ':') { out.push_back({Tok::Op, ":", i}); in_value=true; ++i; continue; }
            if (c == '}') { out.push_back({Tok::Bracket, "}", i}); in_block=false; ++i; continue; }
            int j = i; while (j < n && code[j] != ':' && code[j] != ';' && code[j] != '}' && code.compare(j,2,"/*") != 0) ++j;
            if (j == i) ++j;
            out.push_back({Tok::Prop, code.substr(i, j-i), i}); i=j; continue;
        }
        if (c == ';') { out.push_back({Tok::Op, ";", i}); in_value=false; ++i; continue; }
        if (c == '}') { out.push_back({Tok::Bracket, "}", i}); in_block=false; ++i; continue; }
        if (IsDigit(c)) {
            int j = i + 1; while (j < n && (std::isalnum((unsigned char)code[j]) || code[j] == '.' || code[j] == '%')) ++j;
            out.push_back({Tok::Number, code.substr(i, j-i), i}); i=j; continue;
        }
        if (std::isalpha((unsigned char)c) || c == '-' || c == '_') {
            int j = i + 1; while (j < n && (IsId(code[j]) || code[j] == '-')) ++j;
            out.push_back({Tok::Value, code.substr(i, j-i), i}); i=j; continue;
        }
        out.push_back({Tok::Text, std::string(1, c), i}); ++i;
    }
    return out;
}

std::vector<Tok> TokenizeHtml(const std::string& code) {
    // Simplified — treats tags/attrs/strings/attrs; leaves scripts/styles as text
    std::vector<Tok> out;
    int i = 0, n = (int)code.size();
    while (i < n) {
        if (code.compare(i, 4, "<!--") == 0) {
            int j = (int)code.find("-->", i + 4);
            j = (j == (int)std::string::npos) ? n : j + 3;
            out.push_back({Tok::Comment, code.substr(i, j-i), i}); i=j; continue;
        }
        if (code[i] == '<') {
            int j = i + 1;
            if (j < n && code[j] == '/') ++j;
            int ns = j;
            while (j < n && (std::isalnum((unsigned char)code[j]) || code[j]==':'||code[j]=='-')) ++j;
            out.push_back({Tok::Bracket, "<", i});
            if (i+1<n && code[i+1]=='/') out.push_back({Tok::Bracket, "/", i+1});
            if (j > ns) out.push_back({Tok::Keyword, code.substr(ns, j-ns), ns});
            int k = j;
            while (k < n && code[k] != '>') {
                if (code[k] == '"' || code[k] == '\'') {
                    char q = code[k]; int s = k++;
                    while (k < n && code[k] != q) { if (code[k]=='\\') k++; k++; }
                    if (k < n) k++;
                    out.push_back({Tok::String, code.substr(s, k-s), s});
                } else if (IsWs(code[k])) {
                    int s = k; while (k<n && IsWs(code[k]))++k;
                    out.push_back({Tok::Ws, code.substr(s, k-s), s});
                } else if (code[k] == '=') { out.push_back({Tok::Op, "=", k}); ++k; }
                else {
                    int s = k; while (k<n && !IsWs(code[k]) && code[k]!='=' && code[k]!='>' && code[k]!='/' && code[k]!='"' && code[k]!='\'') ++k;
                    if (k == s) ++k;
                    out.push_back({Tok::Attr, code.substr(s, k-s), s});
                }
            }
            if (k < n) { out.push_back({Tok::Bracket, ">", k}); ++k; }
            i = k; continue;
        }
        int j = (int)code.find('<', i);
        if (j == (int)std::string::npos) j = n;
        out.push_back({Tok::Text, code.substr(i, j-i), i}); i=j;
    }
    return out;
}

std::vector<ColoredSpan> Highlight(const std::string& code, const std::string& lang) {
    std::vector<Tok> toks;
    if (lang == "html") toks = TokenizeHtml(code);
    else if (lang == "css") toks = TokenizeCss(code);
    else toks = TokenizeCode(code, lang);

    std::vector<ColoredSpan> spans;
    spans.reserve(toks.size());
    for (auto& t : toks) spans.push_back({t.value, ColFor(t.type)});
    return spans;
}

std::set<int> ComputeBracketMatch(const std::vector<Tok>& tokens, int caret) {
    int idx = -1;
    for (int i = 0; i < (int)tokens.size(); ++i) {
        if (tokens[i].type == Tok::Bracket &&
            tokens[i].start <= caret && caret <= tokens[i].start + (int)tokens[i].value.size()) {
            idx = i; break;
        }
    }
    if (idx < 0) return {};
    std::string ch = tokens[idx].value;
    std::string pair; int dir;
    if      (ch == "(") { pair = ")"; dir =  1; }
    else if (ch == "[") { pair = "]"; dir =  1; }
    else if (ch == "{") { pair = "}"; dir =  1; }
    else if (ch == ")") { pair = "("; dir = -1; }
    else if (ch == "]") { pair = "["; dir = -1; }
    else if (ch == "}") { pair = "{"; dir = -1; }
    else return {};

    int depth = 0;
    for (int j = idx; j >= 0 && j < (int)tokens.size(); j += dir) {
        if (tokens[j].type != Tok::Bracket) continue;
        if (tokens[j].value == ch)   ++depth;
        else if (tokens[j].value == pair) {
            --depth;
            if (depth == 0) return { tokens[idx].start, tokens[j].start };
        }
    }
    return {};
}

}} // namespace
```
